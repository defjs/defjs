```ts
interface HttpRequest {
  abort?: AbortSignal
  baseEndpoint?: string
  body?: Blob | ArrayBuffer | FormData | URLSearchParams | ReadableStream<Uint8Array> | object | string | number | boolean | null
  bodyContentType?: string | null
  downloadProgress?: HttpProgressFn
  endpoint: string
  headers?: Headers
  method: string
  operation?: string
  queryParams?: URLSearchParams
  queryString?: string
  readonly responseType?: HttpResponseType
  readonly errorResponseType?: HttpResponseType
  timeout?: number
  uploadProgress?: HttpProgressFn
  withCredentials?: boolean
  xsrf?: {
    cookieName: string
    headerName: string
    tokenProvider?: (context: { request: HttpRequest }) => string | null | undefined
  }
  // plus an @internal bodyContentTypeSource field
}
```

الطلب بعد تشغيل الـ builder، كما تستخدمه المعالِجات والمعترضات و`fetchHandler`.

- `body` هو بالفعل القيمة التي تُرسَل على الشبكة. وفي JSON يكون **سلسلة نص JSON**؛ فوقّع HMAC أو الملخّصات (digests) على تلك السلسلة، لا على مدخل الـ struct الأصلي.
- `endpoint` هو المسار بعد ملء معاملات المسار؛ و`baseEndpoint` هو نقطة نهاية العميل.
- يرى المعترضون `queryParams` كـ `new URLSearchParams(queryString)`. والـ builder الذي يُرجع `URLSearchParams` يمرّر تلك النسخة كما هي.
- `operation` هوية ثابتة منخفضة الكاردينالية (low-cardinality) لنقطة النهاية، للقياس عن بُعد (telemetry) والتشخيص.
- `downloadProgress` و`uploadProgress` هما خطّافا `onDownloadProgress` و`onUploadProgress` من `HttpExecuteOptions`.
- `responseType` و`errorResponseType` للقراءة فقط: يُشتقّان من kind الـ struct لـ `output` و`error` في نقطة النهاية ولا يجوز تغييرهما. يُضبط كلٌّ منهما فقط عند إعلان الـ struct الخاص به. والمعترض الذي يختصر المسار بـ `makeResponse` يقرؤهما ليطابق وسيلة النقل.
