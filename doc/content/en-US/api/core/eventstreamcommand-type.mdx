```ts
interface EventStreamCommand<
  TInput extends AnyStruct | undefined,
  TEvents extends EventStructs,

  readonly endpoint: EventStreamDefinition<TInput, TEvents, TError> & { readonly method: string }
  readonly input: EndpointInput<TInput> | undefined
  // plus internal symbol-keyed members that identify and run the command
}
```

Created by calling an `EventStreamEndpoint` with input. Treat it as opaque and pass it to `client.execute`.
