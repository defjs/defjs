```ts
type EventStreamFaultCode = Extract<
  FaultCode,
  | 'CAP_BUFFER_EXCEEDED'
  | 'CAP_QUEUE_OVERFLOW'
  | 'EXT_HOOK_FAILED'
  | 'EXT_OBSERVER_FAILED'
  | 'NET_TIMEOUT'
  | 'NET_UNREACHABLE'
  | 'RES_DECODE_FAILED'
  | 'RES_MEDIA_TYPE_INVALID'
>
```

共用 `FaultCode` 集合的子集。它是 `kind: 'error'` 之 `EventStreamCloseInfo` 的 `code`。
